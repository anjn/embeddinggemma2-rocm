cmake_minimum_required(VERSION 3.22)
project(eg2_llama_adapter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(LLAMA_ROOT "" CACHE PATH "Pinned llama.cpp source checkout")
set(LLAMA_BUILD "${LLAMA_ROOT}/build" CACHE PATH "llama.cpp build directory")
if(NOT EXISTS "${LLAMA_ROOT}/include/llama.h")
 message(FATAL_ERROR "Set LLAMA_ROOT to the pinned llama.cpp checkout")
endif()
find_library(LLAMA_LIBRARY llama PATHS "${LLAMA_BUILD}/bin" NO_DEFAULT_PATH REQUIRED)
find_library(MTMD_LIBRARY mtmd PATHS "${LLAMA_BUILD}/bin" NO_DEFAULT_PATH REQUIRED)
find_library(GGML_LIBRARY ggml PATHS "${LLAMA_BUILD}/bin" NO_DEFAULT_PATH REQUIRED)
find_library(GGML_BASE_LIBRARY ggml-base PATHS "${LLAMA_BUILD}/bin" NO_DEFAULT_PATH REQUIRED)
add_library(eg2_llama_adapter SHARED adapter.cpp)
target_include_directories(eg2_llama_adapter PRIVATE "${LLAMA_ROOT}/include" "${LLAMA_ROOT}/ggml/include" "${LLAMA_ROOT}/tools/mtmd")
target_link_libraries(eg2_llama_adapter PRIVATE ${MTMD_LIBRARY} ${LLAMA_LIBRARY} ${GGML_LIBRARY} ${GGML_BASE_LIBRARY})
set_target_properties(eg2_llama_adapter PROPERTIES BUILD_RPATH "${LLAMA_BUILD}/bin")
