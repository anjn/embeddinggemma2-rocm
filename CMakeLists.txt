cmake_minimum_required(VERSION 3.22)
project(embeddinggemma2_rocm VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(ROCM_ROOT "/opt/rocm" CACHE PATH "ROCm installation")
set(TORCH_ROOT "" CACHE PATH "LibTorch root, e.g. site-packages/torch")
option(EG2_BUILD_NATIVE "Build existing EmbeddingGemma native library" ON)
option(EG2_BUILD_MOBILECLIP2 "Build MobileCLIP2 native library" OFF)
if(EG2_BUILD_MOBILECLIP2)
 include(cmake/MobileCLIP2.cmake)
endif()
if(NOT EG2_BUILD_NATIVE)
 return()
endif()
set(EG2_GPU_ARCH "gfx1150" CACHE STRING "HIP target architecture")
set(EG2_CXX11_ABI 1 CACHE STRING "Must match LibTorch's _GLIBCXX_USE_CXX11_ABI")
if(NOT EXISTS "${TORCH_ROOT}/include/ATen/ATen.h")
 message(FATAL_ERROR "Set TORCH_ROOT to the ROCm LibTorch root")
endif()
set(EG2_DEPS "${CMAKE_BINARY_DIR}/deps")
file(MAKE_DIRECTORY "${EG2_DEPS}/migraphx/api")
file(WRITE "${EG2_DEPS}/migraphx/api/export.h" "#pragma once\n#define MIGRAPHX_C_EXPORT __attribute__((visibility(\"default\")))\n")
if(NOT EXISTS "${EG2_DEPS}/migraphx/migraphx.h")
 file(DOWNLOAD "https://raw.githubusercontent.com/ROCm/AMDMIGraphX/1afd1b89c/src/api/include/migraphx/migraphx.h" "${EG2_DEPS}/migraphx/migraphx.h" TLS_VERIFY ON STATUS status)
 if(NOT status MATCHES "^0;")
  message(FATAL_ERROR "Cannot fetch pinned MIGraphX C header: ${status}")
 endif()
endif()
if(NOT EXISTS "${EG2_DEPS}/json.hpp")
 file(DOWNLOAD "https://raw.githubusercontent.com/nlohmann/json/v3.11.3/single_include/nlohmann/json.hpp" "${EG2_DEPS}/json.hpp" TLS_VERIFY ON STATUS status)
 if(NOT status MATCHES "^0;")
  message(FATAL_ERROR "Cannot fetch pinned JSON header: ${status}")
 endif()
endif()
add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/preprocess.o"
 COMMAND "${ROCM_ROOT}/bin/hipcc" -O3 -fPIC --offload-arch=${EG2_GPU_ARCH} -c "${CMAKE_CURRENT_SOURCE_DIR}/kernels/preprocess.hip" -o "${CMAKE_BINARY_DIR}/preprocess.o"
 DEPENDS kernels/preprocess.hip python/embeddinggemma2_rocm/_compiler/kernels/preprocess.cpp VERBATIM)
add_library(eg2 SHARED src/runtime.cpp "${CMAKE_BINARY_DIR}/preprocess.o")
target_include_directories(eg2 PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE "${EG2_DEPS}" "${TORCH_ROOT}/include" "${TORCH_ROOT}/include/torch/csrc/api/include" "${ROCM_ROOT}/include")
target_compile_definitions(eg2 PRIVATE USE_ROCM=1 __HIP_PLATFORM_AMD__=1 _GLIBCXX_USE_CXX11_ABI=${EG2_CXX11_ABI})
foreach(lib torch torch_cpu torch_hip c10 c10_hip)
 find_library(LIB_${lib} NAMES ${lib} PATHS "${TORCH_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
 target_link_libraries(eg2 PRIVATE ${LIB_${lib}})
endforeach()
set(EG2_TOKENIZER_LIBRARY "" CACHE FILEPATH "Prebuilt tokenizer static library (optional)")
if(NOT EG2_TOKENIZER_LIBRARY)
 find_program(CARGO cargo REQUIRED)
 set(EG2_TOKENIZER_LIBRARY "${CMAKE_BINARY_DIR}/tokenizer/release/libeg2_tokenizer.a")
 add_custom_command(OUTPUT "${EG2_TOKENIZER_LIBRARY}" COMMAND ${CARGO} build --locked --release --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/tokenizer/Cargo.toml" --target-dir "${CMAKE_BINARY_DIR}/tokenizer" DEPENDS tokenizer/Cargo.toml tokenizer/Cargo.lock tokenizer/src/lib.rs VERBATIM)
 add_custom_target(eg2-tokenizer-build DEPENDS "${EG2_TOKENIZER_LIBRARY}")
 add_dependencies(eg2 eg2-tokenizer-build)
endif()
target_link_libraries(eg2 PRIVATE "${EG2_TOKENIZER_LIBRARY}" dl pthread m)
find_library(MIGRAPHX_C NAMES migraphx_c libmigraphx_c.so.3 PATHS "${ROCM_ROOT}/lib" REQUIRED)
find_library(HIP_RUNTIME NAMES amdhip64 PATHS "${ROCM_ROOT}/lib" REQUIRED)
target_link_libraries(eg2 PRIVATE ${MIGRAPHX_C} ${HIP_RUNTIME})
# ROCm PyTorch wheels name HIP both libamdhip64.so and SONAME .so.7.
# Resolve both names to the same inode; loading wheel + system HIP crashes at teardown.
set(EG2_HIP_SHIM "${CMAKE_BINARY_DIR}/hip-runtime")
file(MAKE_DIRECTORY "${EG2_HIP_SHIM}")
if(EXISTS "${TORCH_ROOT}/lib/libamdhip64.so")
 foreach(alias libamdhip64.so libamdhip64.so.7)
  file(CREATE_LINK "${TORCH_ROOT}/lib/libamdhip64.so" "${EG2_HIP_SHIM}/${alias}" SYMBOLIC)
 endforeach()
endif()
# The HIP shim must carry its wheel-matched HSA/compiler dependencies too.
# Otherwise $ORIGIN resolves inside the shim and silently picks system versions.
foreach(pair "libhsa-runtime64.so|libhsa-runtime64.so.1" "libamd_comgr.so|libamd_comgr.so.3" "librocprofiler-register.so|librocprofiler-register.so")
 string(REPLACE "|" ";" aliases "${pair}")
 list(GET aliases 0 source)
 if(EXISTS "${TORCH_ROOT}/lib/${source}")
  foreach(alias IN LISTS aliases)
   file(CREATE_LINK "${TORCH_ROOT}/lib/${source}" "${EG2_HIP_SHIM}/${alias}" SYMBOLIC)
  endforeach()
 endif()
endforeach()
set_target_properties(eg2 PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0 CXX_VISIBILITY_PRESET hidden BUILD_RPATH "${EG2_HIP_SHIM};${TORCH_ROOT}/lib;${ROCM_ROOT}/lib" INSTALL_RPATH "$ORIGIN/eg2-hip;${TORCH_ROOT}/lib;${ROCM_ROOT}/lib")
add_executable(eg2-example examples/embed.cpp)
target_link_libraries(eg2-example PRIVATE eg2)
add_executable(eg2-example-c examples/embed.c)
target_link_libraries(eg2-example-c PRIVATE eg2)
add_executable(eg2-text-example examples/text.cpp)
target_link_libraries(eg2-text-example PRIVATE eg2)
set_target_properties(eg2-example eg2-example-c eg2-text-example PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
install(TARGETS eg2 eg2-example eg2-example-c eg2-text-example LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})

install(DIRECTORY "${EG2_HIP_SHIM}/" DESTINATION "${CMAKE_INSTALL_LIBDIR}/eg2-hip")

install(FILES LICENSE THIRD_PARTY_NOTICES.md THIRD_PARTY_LICENSES.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/eg2)
